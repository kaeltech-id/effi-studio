**Task 2 — Add version sync to script** `M`
- File: `studio/scripts/build-macos-dmg.sh` (append)

- Reuse: `jq` (confirmed present) for reading back, `python3` for the write — the repo already depends on Python 3.8+
- Code sample:
  ```bash
  echo "==> syncing version into tauri.conf.json and Cargo.toml"
  TAURI_CONF="$STUDIO_DIR/src-tauri/tauri.conf.json"
  CARGO_TOML="$STUDIO_DIR/src-tauri/Cargo.toml"

  VERSION="$VERSION" python3 -I - "$TAURI_CONF" <<'PY'
  import json, os, sys
  path = sys.argv[1]
  with open(path) as fh:
      conf = json.load(fh)
  conf["version"] = os.environ["VERSION"]
  with open(path, "w") as fh:
      json.dump(conf, fh, indent=2)
      fh.write("\n")
  PY

  python3 -I -c "import json,sys; json.load(open(sys.argv[1]))" "$TAURI_CONF"

  VERSION="$VERSION" python3 -I - "$CARGO_TOML" <<'PY'
  import os, re, sys
  path = sys.argv[1]
  with open(path) as fh:
      body = fh.read()
  body, n = re.subn(r'^version = "[^"]*"$',
                    'version = "%s"' % os.environ["VERSION"],
                    body, count=1, flags=re.M)
  if n != 1:
      sys.exit("error: no [package] version line found in %s" % path)
  with open(path, "w") as fh:
      fh.write(body)
  PY

  grep -q "^version = \"$VERSION\"\$" "$CARGO_TOML" || { echo "error: Cargo.toml version not synced" >&2; exit 1; }
  echo "    tauri.conf.json → $(jq -r .version "$TAURI_CONF")"
  ```

- Edge cases: malformed JSON on entry → `json.load` raises and `set -e` aborts before any build; sed failing to match → explicit `grep -q` guard exits 1 rather than building a mislabelled DMG

  ```bash
  echo "==> verifying $STABLE_NAME"
  MOUNT_POINT=$(/usr/bin/mktemp -d "${TMPDIR:-/tmp}/effi-dmg.XXXXXX")
  cleanup() { /usr/bin/hdiutil detach "$MOUNT_POINT" -quiet 2>/dev/null || true; /bin/rm -rf "$MOUNT_POINT"; }
  trap cleanup EXIT
  /usr/bin/hdiutil attach "$DIST_DIR/$STABLE_NAME" -nobrowse -readonly -mountpoint "$MOUNT_POINT" -quiet

  APP="$MOUNT_POINT/Effi Studio.app"
  [ -d "$APP" ] || { echo "error: 'Effi Studio.app' missing inside the DMG" >&2; exit 1; }
  BIN="$APP/Contents/MacOS/Effi Studio"
  [ -f "$BIN" ] || BIN=$(/usr/bin/find "$APP/Contents/MacOS" -maxdepth 1 -type f | head -1)
  echo "    archs: $(/usr/bin/lipo -archs "$BIN" 2>/dev/null || echo 'unknown')"
  echo "    bundle version: $(/usr/bin/defaults read "$APP/Contents/Info.plist" CFBundleShortVersionString 2>/dev/null || echo 'unknown')"
  cleanup
  trap - EXIT
  ```

**Task 14 — Add release-asset guard test** `M`
- File: `tests/test_release_assets.py` (create)
- Reuse: `tests/test_studio_guards.py` conventions — a `_repo(*parts)` helper and a class docstring naming the concrete failure the guard prevents
- Code sample:
  ```python
  import json
  import os
  import re

  STABLE_ASSET = 'EffiStudio-macOS.dmg'
  DOWNLOAD_URL = (
      'https://github.com/kaeltech-id/effi-studio/releases/latest/download/'
      + STABLE_ASSET)


  def _repo(*parts):
      return os.path.join(os.path.dirname(__file__), '..', *parts)


  def _read(*parts):
      with open(_repo(*parts), encoding='utf-8') as fh:
          return fh.read()


  class TestDownloadLinkMatchesBuiltAsset:
      """The README download link is a bare URL: nothing fails if the build
      script renames its output, the link just 404s for every visitor. These
      guards tie the link, the script and the workflow to one filename."""

      def test_readme_links_the_stable_asset(self):
          assert DOWNLOAD_URL in _read('README.md')

      def test_script_produces_that_exact_name(self):
          script = _read('studio', 'scripts', 'build-macos-dmg.sh')
          assert f'STABLE_NAME="{STABLE_ASSET}"' in script

      def test_workflow_uploads_that_exact_name(self):
          assert STABLE_ASSET in _read('.github', 'workflows', 'release-macos.yml')

      def test_workflow_calls_the_script_instead_of_rebuilding(self):
          workflow = _read('.github', 'workflows', 'release-macos.yml')
          assert 'studio/scripts/build-macos-dmg.sh' in workflow
          assert 'tauri -- build' not in workflow

      def test_script_never_runs_git_or_gh(self):
          script = _read('studio', 'scripts', 'build-macos-dmg.sh')
          for line in script.splitlines():
              stripped = line.strip()
              assert not re.match(r'^(gh release create|git (tag|push|commit))\b',
                                  stripped), f'script executes: {stripped}'


  class TestVersionSourcesAgree:
      """VERSION, tauri.conf.json and Cargo.toml all feed the DMG filename. A
      drift between them ships an installer labelled with the wrong release."""

      def _version(self):
          return _read('VERSION').strip()

      def test_tauri_conf_matches_version_file(self):
          conf = json.loads(_read('studio', 'src-tauri', 'tauri.conf.json'))
          assert conf['version'] == self._version()

      def test_cargo_toml_matches_version_file(self):
          cargo = _read('studio', 'src-tauri', 'Cargo.toml')
          found = re.search(r'^version = "([^"]+)"', cargo, re.M)
          assert found and found.group(1) == self._version()


  class TestQuarantineInstructionPresent:
      """An ad-hoc signed DMG downloaded from the internet is refused by
      Gatekeeper as 'damaged'. Without this line in the README the download is
      useless to a first-time user."""

      def test_readme_documents_the_quarantine_fix(self):
          assert 'xattr -dr com.apple.quarantine' in _read('README.md')
  ```
